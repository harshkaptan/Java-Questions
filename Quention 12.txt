class Main {
    public static void main(String[] args) {
        String str = "HarsH";
        char[] ch = str.toCharArray(); 
        for (int i = 0; i < ch.length; i++) {
            if (ch[i] >= 'A' && ch[i] <= 'Z') {
                ch[i] = (char)(ch[i] + 32);  
            }
        }
        String result = "";
        for (int i = 0; i < ch.length; i++) {
            result += ch[i];
        }

        System.out.println(result);  
    }
}
