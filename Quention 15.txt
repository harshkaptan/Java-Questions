public class FirstOccurrenceBinarySearch {
    public static int FirstOccurrence(int[] arr, int target) {
        int low = 0, high = arr.length - 1;
        int result = -1;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (arr[mid] == target) {
                result = mid;
                high = mid - 1;
            }  if (arr[mid] < target) {
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }

        return result;
    }

    public static void main(String[] args) {
        int[] arr = {2, 3, 4, 4, 4, 5, 6, 6, 7};
        int target = 4;

        System.out.println(FirstOccurrence(arr, target));
    }
}
