class Main {
    public static void main(String[] args) {
        int[] arr = {2,7,5,4,8,6};
        int target = 6;
        System.out.println(LinearSearch(arr,target));
    }
    static int LinearSearch(int[] arr, int target)
    {
        for (int i=0;i<arr.length;i++){
            if(arr [i]==target)
            {
                return i;
            }
        }
        return - 1;
    }
}
