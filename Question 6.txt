class Main {
    public static void main(String[] args) {
        String str1 = "my car is red";
        String newstr = "";
        
        for (int i = 0; i < str1.length(); i++) {
            if (str1.charAt(i) == ' ') {
                continue;
            } else {
                newstr = newstr + str1.charAt(i);
            }
        }

        System.out.println(newstr); 
    }
}
