class Main {
    public static void main(String[] args) {
        int[] arr = {2, 7, 3, 4, 3, 6, 3};
        int target = 3;
        System.out.println("Frequency of " + target + " is: " + frequency(arr, target));
    }

    static int frequency(int[] arr, int target) {
        int count = 0;
        for (int i = 0; i < arr.length; i++) { 
            if (arr[i] == target) {
                count++;
            }
        }
        return count;
    }
}
